#include <stdio.stdio.h>

// Global variable
int global_count = 10;

void count_function() {
    // Static local variable
    static int calls = 0;
    calls++;
    printf("Function called %d time(s)\n", calls);
}

int main() {
    // Local variable
    int user_age = 20;
    
    printf("Global count: %d\n", global_count);
    printf("User age: %d\n", user_age);
    
    count_function(); // Output: Function called 1 time(s)
    count_function(); // Output: Function called 2 time(s)

    return 0;
}
