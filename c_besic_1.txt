if / else-if / else
________________________________________________________________________________________________________
                        ┌────────────────────────┐
                        │    START EVALUATION    │
                        └───────────┬────────────┘
                                    │
                         ┌──────────┴──────────┐
                         │  Is Condition 1?    ├───────┐ YES
                         └──────────┬──────────┘       │
                                    │ NO               ▼
                         ┌──────────┴──────────┐ ┌───────────────┐
                         │  Is Condition 2?    ├─┤  if Block     │
                         └──────────┬──────────┘ │ (Executes 1)  │
                                    │ NO         └───────┬───────┘
                                    ▼                    │
                         ┌─────────────────────┐         │
                         │     else Block      │         │
                         │ (If all rules fail) │         │
                         └──────────┬──────────┘         │
                                    │                    │
                                    └──────────┬─────────┘
                                               ▼
                        ┌────────────────────────┐
                        │  CONTINUE PROGRAM FLOW │
                        └────────────────────────┘



#include <stdio.h>

int main() {
    int temperature = 28;

    if (temperature > 35) {
        printf("Weather is hot.\n");
    } 
    else if (temperature >= 20) {
        printf("Weather is warm.\n"); // This evaluates to true; executes and exits ladder
    } 
    else if (temperature >= 10) {
        printf("Weather is cool.\n"); // Skipped entirely
    } 
    else {
        printf("Weather is cold.\n"); // Skipped entirely
    }

    return 0;
}

Nested if-else
________________________________________________________________________________________________________
                        ┌────────────────────────┐
                        │    START EVALUATION    │
                        └───────────┬────────────┘
                                    │
                         ┌──────────┴──────────┐
                         │   Condition 1?      ├───────┐ NO (Go to Else/Ladder)
                         └──────────┬──────────┘       │
                                    │ YES              ▼
                         ┌──────────┴──────────┐ ┌───────────────┐
                         │   Condition 2?      │ │   Outer else  │
                         │ (Nested/Inner Check)│ │  (or Ladder)  │
                         └────┬───────────┬────┘ └───────┬───────┘
                          YES │           │ NO           │
                              ▼           ▼              │
                      ┌───────────┐   ┌───────────┐      │
                      │Inner Code │   │Inner else │      │
                      │ Block 1   │   │ Block 2   │      │
                      └─────┬─────┘   └─────┬─────┘      │
                            │               │            │
                            └───────┬───────┴────────────┘
                                    ▼
                        ┌────────────────────────┐
                        │  CONTINUE PROGRAM FLOW │
                        └────────────────────────┘

#include <stdio.h>

int main() {
    int age = 22;
    int weight = 45;

    // Outer Primary Condition
    if (age >= 18) {
        printf("Primary Check Passed: You are an adult.\n");

        // Inner Nested Condition
        if (weight >= 50) {
            printf("Success: You are eligible to donate blood.\n");
        } 
        else {
            printf("Rejected: You meet the age limit, but your weight is too low.\n");
        }
    } 
    else {
        printf("Rejected: You must be at least 18 years old.\n");
    }

    return 0;
}

switch statement
________________________________________________________________________________________________________
                        ┌────────────────────────┐
                        │    START EVALUATION    │
                        │    (Expression/Var)    │
                        └───────────┬────────────┘
                                    │
         ┌──────────────────────────┼──────────────────────────┐
         │ Match?                   │ Match?                   │ Match?
         ▼                          ▼                          ▼
 ┌───────────────┐          ┌───────────────┐          ┌───────────────┐
 │    case 1     │          │    case 2     │          │    default    │
 ├───────────────┤          ├───────────────┤          ├───────────────┤
 │ Execute Code  │          │ Execute Code  │          │ Execute Code  │
 ├───────────────┤          ├───────────────┤          └───────┬───────┘
 │ break; ───────┼─┐        │ break; ───────┼─┐                │
 └───────────────┘ │        └───────────────┘ │                │
                   │                          │                │
                   └─────────────────┬────────┴────────────────┘
                                     ▼
                        ┌────────────────────────┐
                        │  CONTINUE PROGRAM FLOW │
                        └────────────────────────┘


#include <stdio.h>

int main() {
    char grade = 'B';

    switch (grade) {
        case 'A':
            printf("Excellent performance!\n");
            break; // Exits the switch block immediately
            
        case 'B':
            printf("Good job!\n");
            break; // Matches here, executes, and exits
            
        case 'C':
            printf("You passed.\n");
            break;
            
        default:
            printf("Invalid grade entered.\n");
            // No break needed for the final default case
    }

    return 0;
}

for loop
________________________________________________________________________________________________________
                        ┌────────────────────────┐
                        │    1. INITIALIZATION   │ (Runs exactly once)
                        └───────────┬────────────┘
                                    │
                                    ▼
                        ┌────────────────────────┐
                 ┌─────>│    2. TEST CONDITION   │
                 │      └───────────┬────────────┘
                 │                  │
                 │         Is condition True?
                 │         ┌────────┴────────┐
                 │     YES │                 │ NO
                 │         ▼                 ▼
                 │   ┌───────────────┐ ┌───────────────┐
                 │   │ 3. LOOP BODY  │ │ EXIT LOOP     │
                 │   │ (Code Blocks) │ │ (Move Past)   │
                 │   └──────┬────────┘ └───────────────┘
                 │          │
                 │          ▼
                 │   ┌───────────────┐
                 └───│ 4. ITERATION  │
                     │ (Increment/Dec│
                     └───────────────┘

#include <stdio.h>

int main() {
    // Structure: for (initialization; condition; increment/decrement)
    for (int i = 1; i <= 5; i++) {
        printf("Iteration Number: %d\n", i);
    }
    
    printf("Loop complete!\n");
    return 0;
}

while loop
________________________________________________________________________________________________________
                        ┌────────────────────────┐
                        │ 1. LOOP INITIALIZATION │ (Set variable before loop)
                        └───────────┬────────────┘
                                    │
                                    ▼
                        ┌────────────────────────┐
                 ┌─────>│   2. CHECK CONDITION   │
                 │      └───────────┬────────────┘
                 │                  │
                 │         Is condition True?
                 │         ┌────────┴────────┐
                 │     YES │                 │ NO
                 │         ▼                 ▼
                 │   ┌───────────────┐ ┌───────────────┐
                 │   │ 3. LOOP BODY  │ │  EXIT LOOP    │
                 │   │ (Code Blocks) │ │ (Move Past)   │
                 │   └──────┬────────┘ └───────────────┘
                 │          │
                 │          ▼
                 │   ┌───────────────┐
                 └───│ 4. UPDATE VAR │ (Increment/Decrement)
                     └───────────────┘


#include <stdio.h>

int main() {
    int i = 1; // 1. Initialization

    // 2. Check Condition
    while (i <= 5) {
        printf("Count: %d\n", i); // 3. Loop Body

        i++; // 4. Update Variable (Crucial to prevent infinite loop)
    }

    printf("Loop complete!\n");
    return 0;
}

do-while loop
________________________________________________________________________________________________________
                        ┌────────────────────────┐
                        │ 1. LOOP INITIALIZATION │ (Set variable before loop)
                        └───────────┬────────────┘
                                    │
                                    ▼
                             ┌───────────────┐
                 ┌──────────>│ 2. LOOP BODY  │
                 │           │ (Code Blocks) │
                 │           └──────┬────────┘
                 │                  │
                 │                  ▼
                 │           ┌───────────────┐
                 │           │ 3. UPDATE VAR │ (Increment/Decrement)
                 │           └──────┬────────┘
                 │                  │
                 │                  ▼
                 │      ┌────────────────────────┐
                 │      │   4. CHECK CONDITION   │
                 └──────┤  (Evaluated at bottom) │
                        └───────────┬────────────┘
                                    │
                           Is condition True?
                           ┌────────┴────────┐
                       YES │                 │ NO
                           ▼                 ▼
                   ┌───────────────┐ ┌───────────────┐
                   │  Loop Back    │ │   EXIT LOOP   │
                   └───────────────┘ └───────────────┘

#include <stdio.h>

int main() {
    int num;

    do {
        printf("Enter a number (0 to quit): ");
        scanf("%d", &num); // 2. Loop Body (Runs at least once)
        
        printf("You entered: %d\n", num);
        
    } while (num != 0); // 4. Check Condition (Note the trailing semicolon)

    printf("Loop complete!\n");
    return 0;
}

Break
________________________________________________________________________________________________________
    [A] Start Loop
         │
         ▼
 ┌──[B] Loop Condition Valid?
 │       │
 │       ├─► (Yes) ──► [C] Check Break Condition
 │       │                  │
 │       │                  ├─► (True)  ──► [D] Execute 'break' ──┐
 │       │                  │                                     │
 │       │                  └─► (False) ──► [E] Run Loop Body ┘   │
 │       │                                       │                │
 │       └◄──────────────────────────────────────┘                │
 │                                                                │
 └──────► (No / Broken) ──► [F] Loop Terminated ◄─────────────────┘
                             (Code Outside Loop)


#include <stdio.h>

int main() {
    for (int i = 1; i <= 5; i++) {
        if (i == 3) {
            break; // Exits the loop entirely when i equals 3
        }
        printf("%d ", i);
    }
    // Output: 1 2
    return 0;
}

continue
________________________________________________________________________________________________________
    [A] Start Loop
         │
         ▼
 ┌──[B] Loop Condition Valid? ◄──────────────────────────────┐
 │       │                                                   │
 │       ├─► (Yes) ──► [C] Check Continue Condition           │
 │       │                  │                                │
 │       │                  ├─► (True)  ──► [D] 'continue' ──┤
 │       │                  │                                │
 │       │                  └─► (False) ──► [E] Run Loop Body│
 │       │                                       │           │
 │       └───────────────────────────────────────┴───────────┘
 │
 └──────► (No) ───────────► [F] Loop Terminated 
                             (Code Outside Loop)

#include <stdio.h>

int main() {
    for (int i = 1; i <= 5; i++) {
        if (i == 3) {
            continue; // Skips printing 3 and jumps straight to i++
        }
        printf("%d ", i);
    }
    // Output: 1 2 4 5
    return 0;
}
